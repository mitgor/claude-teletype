Update `_kill_process()` to use backend's proc_holder if available (for Claude CLI), or be a no-op for API backends:
```python
async def _kill_process(self) -> None:
    if self._backend is not None and hasattr(self._backend, 'proc_holder'):
        proc_holder = self._backend.proc_holder
        if not proc_holder:
            return
        proc = proc_holder[0]
        if proc.returncode is not None:
            proc_holder.clear()
            return
        proc.terminate()
        try:
            await asyncio.wait_for(proc.wait(), timeout=5.0)
        except asyncio.TimeoutError:
            proc.kill()
            await proc.wait()
        proc_holder.clear()
```
